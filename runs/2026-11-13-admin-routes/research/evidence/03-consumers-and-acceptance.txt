export function isVanityAvailable(
	vanitySettings: VanitySettings<unknown> | null | undefined,
	vanityName: string,
	now = nowInSeconds(),
): boolean {
	if (!vanityName) return false
	if (!vanitySettings || !vanitySettings.entries) return true
	const existing = vanitySettings.entries.find((e) => e.vanityName === vanityName.toLowerCase())
	if (!existing) return true
	return existing.validUntil < now
}

export function resolveVanity(
	vanitySettings: VanitySettings<unknown> | null | undefined,
	vanityName: string,
	now = nowInSeconds(),
): VanityEntry | null {
	if (!vanitySettings || !vanitySettings.entries || !vanityName) return null
	const entry = vanitySettings.entries.find((e) => e.vanityName === vanityName.toLowerCase())
	if (!entry || entry.validUntil < now) return null
	return entry
}

export function StorefrontAddressPage({ dTag }: { dTag: string }) {
	const { data: config, isLoading: isLoadingConfig } = useConfigQuery()
	const author = config?.appPublicKey ?? ''
	const hasAddress = isValidHexKey(author) && dTag.length > 0
	const { data: page, isLoading: isLoadingPage, isError, refetch } = useStorefrontAddressPage(hasAddress ? author : undefined, dTag)

	// The config read gates the page read (the query stays disabled without a
	// valid author), so the two loads are strictly sequential: wait for the
	// config first, then for the page.
	if (isLoadingConfig || (hasAddress && isLoadingPage)) {
		return <LoadingState className="min-h-screen" label="Loading…" />
	}

	if (isError) {
		return (
			<ErrorState
				className="min-h-screen"
				title="This page could not be loaded"
				message="The relay holding this page did not respond. Trying again may help."
				onRetry={() => void refetch()}
				retryTestId="storefront-address-retry"
			/>
		)
	}

	if (shouldRenderStorefrontPage(page) && page.pubkey) {
		return <StorefrontRenderer page={page.page} pubkey={page.pubkey} slots={storefrontBlockSlots} />
	}

	// Absent, disabled, zero-block and unparseable pages all land here: a
	// deliberate, token-clean empty state, never a blank page or an error.
	return (
		<EmptyState
			className="min-h-screen"
			data-testid="storefront-address-empty"
			title="Nothing published yet"
			description="No page has been published at this address."
		/>
	)
}

export function StorefrontOrProfilePage({ profileId }: { profileId: string }) {
	const { user } = useStore(authStore)
	const isDraftPreview = typeof window !== 'undefined' && new URLSearchParams(window.location.search).get('preview') === 'draft'
	// npub, nprofile and NIP-05 identifiers all have to become a hex pubkey
	// before a page can be addressed. Hex skips the lookup; everything else
	// borrows the profile query `ProfilePage` is about to run anyway, so the
	// resolution costs no extra relay round trip.
	const needsResolution = !isValidHexKey(profileId)
	const profileOptions = profileByIdentifierQueryOptions(profileId)
	const { data: profileData, isLoading: isResolvingProfile } = useQuery({
		...profileOptions,
		enabled: profileOptions.enabled && needsResolution,
	})

	const pubkey = needsResolution ? (profileData?.user?.pubkey ?? '') : profileId

	const { data: storefront, isLoading: isLoadingStorefront } = useStorefrontPage(pubkey)
	const isAuthorPreview = isDraftPreview && isValidHexKey(pubkey) && user?.pubkey === pubkey
	const previewPage = isAuthorPreview ? (loadStorefrontDraft(pubkey) ?? storefront?.page ?? null) : null

	// The preview URL is intentionally local to the signed-in author: it reads
	// their validated local draft (or their stored page when no draft exists),
	// without publishing or exposing that draft to other visitors.
	if (previewPage && previewPage.content.length > 0) {
		return <StorefrontRenderer page={previewPage} pubkey={pubkey} slots={storefrontBlockSlots} />
	}

	// Both queries are in flight concurrently, so this waits on the slower of
	// two requests that would have happened regardless. Deciding before they
	// settle would flash the profile page and then replace it, which reads as a
	// bug to the visitor and undoes the author's first impression.
	const isDeciding = (needsResolution && isResolvingProfile) || (pubkey.length > 0 && isLoadingStorefront)

	if (isDeciding) {
		return <LoadingState className="min-h-screen" label="Loading…" />
	}

	if (shouldRenderStorefrontPage(storefront) && storefront.pubkey) {
		return <StorefrontRenderer page={storefront.page} pubkey={storefront.pubkey} slots={storefrontBlockSlots} />
	}

	return <ProfilePage profileId={profileId} />
}

export async function fetchStorefrontPage(pubkey: string, dTag: string = STOREFRONT_PAGE_D_TAG): Promise<StorefrontPageResult> {
	if (!isValidHexKey(pubkey) || dTag.length === 0) return EMPTY_RESULT

	// Read-seam justification (ADR-READ-SEAM-STREAMS §8): a single mounted page is a
	// single-entity/detail read — one addressable coordinate (author, kind, `d`) — so a
	// bounded `fetch` is the doctrine-sanctioned class here, not a stream.
	const events = await getReadContext().fetch([
		{
			kinds: [STOREFRONT_PAGE_KIND],
			authors: [pubkey],
			'#d': [dTag],
		},
	])

	// Defence in depth: a relay may answer a filter with events that do not
	// match it. Author and `d` decide the address, so re-check both rather than
	// rendering whatever came back.
	const addressed = events.filter(
		(event) =>
			event.kind === STOREFRONT_PAGE_KIND && event.pubkey === pubkey && event.tags.some((tag) => tag[0] === 'd' && tag[1] === dTag),
	)

	const event = selectReplaceable(addressed)
	if (!event) return EMPTY_RESULT

	const parsed = parseStorefrontPage(event.content)
	return { event, pubkey: event.pubkey, ...parsed }
}

/**
 * True when this author's page should replace the default profile view.
 *
 * Both conditions matter: the document must have parsed, and the author must
 * have switched it on. A page that parsed to zero surviving blocks is also
 * refused — an empty page is worse for a visitor than the profile they expected.
 */
export function shouldRenderStorefrontPage(result: StorefrontPageResult | undefined): result is StorefrontPageResult & {
	page: StorefrontPage
} {
	return Boolean(result?.page?.enabled && result.page.content.length > 0)
}
